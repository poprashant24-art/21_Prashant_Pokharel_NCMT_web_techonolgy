    /* ===================== Creative Cursor FX ===================== */
    const lastPointer = { x: innerWidth/2, y: innerHeight/2 };

    function initCreativeCursor(){
      const cursor = document.getElementById("inkCursor");
      const label  = document.getElementById("cursorLabel");
      const canvas = document.getElementById("cursorFx");
      const ctx = canvas.getContext("2d", { alpha: true });

      function resize(){
        const dpr = Math.max(1, Math.min(2, devicePixelRatio || 1));
        canvas.width = Math.floor(innerWidth * dpr);
        canvas.height = Math.floor(innerHeight * dpr);
        canvas.style.width = innerWidth + "px";
        canvas.style.height = innerHeight + "px";
        ctx.setTransform(dpr,0,0,dpr,0,0);
      }
      addEventListener("resize", resize);
      resize();

      let x = lastPointer.x, y = lastPointer.y;
      let cx = x, cy = y, vx = 0, vy = 0;
      const stiffness = 0.18;
      const damping = 0.70;

      let visible = false;
      function show(){
        if(visible) return;
        visible = true;
        cursor.style.opacity = "1";
      }

      const particles = [];
      function addParticle(px, py, velx, vely, life, size){
        particles.push({ x:px, y:py, vx:velx, vy:vely, life, ttl:life, size });
      }

      function step(){
        const dx = x - cx, dy = y - cy;
        vx = (vx + dx * stiffness) * damping;
        vy = (vy + dy * stiffness) * damping;
        cx += vx; cy += vy;

        cursor.style.transform = `translate3d(${cx}px, ${cy}px, 0) translate3d(-50%,-50%,0)`;
        label.style.left = cx + "px";
        label.style.top  = cy + "px";

        ctx.clearRect(0,0,innerWidth,innerHeight);

        for(let i=particles.length-1; i>=0; i--){
          const p = particles[i];
          p.life -= 1;
          p.vx *= 0.98; p.vy *= 0.98;
          p.x += p.vx; p.y += p.vy;

          const a = Math.max(0, p.life / p.ttl);
          const g = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size * 2.4);
          g.addColorStop(0, `rgba(124,92,255,${0.18*a})`);
          g.addColorStop(1, `rgba(0,212,255,${0.05*a})`);
          ctx.fillStyle = g;
          ctx.beginPath();
          ctx.arc(p.x, p.y, p.size * (0.8 + (1-a)*0.6), 0, Math.PI*2);
          ctx.fill();

          if(p.life <= 0) particles.splice(i,1);
        }

        requestAnimationFrame(step);
      }
      step();

      let trailTimer = 0;
      addEventListener("pointermove", (e) => {
        x = e.clientX; y = e.clientY;
        lastPointer.x = x; lastPointer.y = y;
        show();

        trailTimer++;
        if(trailTimer % 2 === 0){
          const speed = Math.hypot(e.movementX, e.movementY);
          const amt = speed > 14 ? 2 : 1;
          for(let i=0;i<amt;i++){
            const jx = (Math.random()-0.5)*4;
            const jy = (Math.random()-0.5)*4;
            addParticle(x + jx, y + jy, (Math.random()-0.5)*0.8, (Math.random()-0.5)*0.8, 26, 2.2);
          }
        }
      });

      const hoverSel = "a, button, .btn, .pill, .chip, .proj, .cap, .x, .role";
      addEventListener("pointerover", (e) => {
        const h = e.target.closest(hoverSel);

        document.body.classList.toggle("cursor-text", !!isText);

        for(let i=0;i<count;i++){
          const ang = Math.random() * Math.PI * 2;
          const spd = 1.2 + Math.random() * 2.2;
          addParticle(px, py,
            Math.cos(ang)*spd,
            Math.sin(ang)*spd,
            34 + Math.floor(Math.random()*12),
            2.4 + Math.random()*1.8
          );
        }
      }
